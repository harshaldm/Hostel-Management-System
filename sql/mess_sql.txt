CREATE DATABASE Hostel_Management;
USE Hostel_Management;

-- Full Mess Management Module
-- 1 Settings
CREATE TABLE IF NOT EXISTS Mess_Settings(setting_name VARCHAR(100) PRIMARY KEY, value DECIMAL(10,2) NOT NULL);
INSERT INTO Mess_Settings VALUES ('refund_per_day',100) ON DUPLICATE KEY UPDATE value=value;

-- 2 Fees
CREATE TABLE IF NOT EXISTS Mess_Fees(
 fee_id INT AUTO_INCREMENT PRIMARY KEY,
 bt_no VARCHAR(20), month INT, year INT, amount DECIMAL(10,2), payment_date DATE, status VARCHAR(20) DEFAULT 'Pending',
 FOREIGN KEY(bt_no) REFERENCES Students(bt_no) ON DELETE CASCADE);

-- 3 Leaves
CREATE TABLE IF NOT EXISTS Mess_Leaves(
 leave_id INT AUTO_INCREMENT PRIMARY KEY,
 bt_no VARCHAR(20), from_date DATE, to_date DATE, days INT,
 FOREIGN KEY(bt_no) REFERENCES Students(bt_no) ON DELETE CASCADE);

-- 4 Refunds
CREATE TABLE IF NOT EXISTS Mess_Refunds(
 refund_id INT AUTO_INCREMENT PRIMARY KEY,
 bt_no VARCHAR(20) UNIQUE,
 total_days INT DEFAULT 0,
 refund_amount DECIMAL(10,2) DEFAULT 0,
 last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
 FOREIGN KEY(bt_no) REFERENCES Students(bt_no) ON DELETE CASCADE);

-- 5 Weekly Menu
CREATE TABLE IF NOT EXISTS Weekly_Menu(
 menu_id INT AUTO_INCREMENT PRIMARY KEY,
 day_name VARCHAR(20), meal_type VARCHAR(20), dish_id INT,
 FOREIGN KEY(dish_id) REFERENCES Menu_Dishes(dish_id));

-- 6 Inventory
CREATE TABLE IF NOT EXISTS Inventory(
 item_id INT AUTO_INCREMENT PRIMARY KEY,
 item_name VARCHAR(100) UNIQUE,
 quantity DECIMAL(10,2), unit VARCHAR(20), cost_per_unit DECIMAL(10,2), total_cost DECIMAL(10,2), week_no INT);

-- 7 Roles
CREATE TABLE IF NOT EXISTS Staff_Roles(
 role_id INT AUTO_INCREMENT PRIMARY KEY,
 role_name VARCHAR(50) UNIQUE,
 salary_per_person DECIMAL(10,2));

-- 8 Staff
CREATE TABLE IF NOT EXISTS Mess_Staff(
 staff_id INT AUTO_INCREMENT PRIMARY KEY,
 name VARCHAR(255), role_id INT, phone VARCHAR(15), joining_date DATE, status VARCHAR(20) DEFAULT 'Active',
 FOREIGN KEY(role_id) REFERENCES Staff_Roles(role_id));

-- 9 Expenses
CREATE TABLE IF NOT EXISTS Mess_Expenses(
 expense_id INT AUTO_INCREMENT PRIMARY KEY,
 type VARCHAR(100), amount DECIMAL(10,2), expense_date DATE, remarks VARCHAR(255));

-- 10 Dishes
CREATE TABLE IF NOT EXISTS Menu_Dishes(
 dish_id INT AUTO_INCREMENT PRIMARY KEY,
 dish_name VARCHAR(100) UNIQUE, meal_type VARCHAR(20));

-- 11 Requirements
CREATE TABLE IF NOT EXISTS Dish_Requirements(
 req_id INT AUTO_INCREMENT PRIMARY KEY,
 dish_id INT, item_id INT, required_qty DECIMAL(10,2),
 UNIQUE(dish_id,item_id),
 FOREIGN KEY(dish_id) REFERENCES Menu_Dishes(dish_id) ON DELETE CASCADE,
 FOREIGN KEY(item_id) REFERENCES Inventory(item_id) ON DELETE CASCADE);

-- 12 Meal Log
CREATE TABLE IF NOT EXISTS Meal_Log(
 log_id INT AUTO_INCREMENT PRIMARY KEY,
 dish_id INT, served_date DATE DEFAULT (CURRENT_DATE), meal_type VARCHAR(20), remarks VARCHAR(255),
 FOREIGN KEY(dish_id) REFERENCES Menu_Dishes(dish_id));

DELIMITER //
DROP TRIGGER IF EXISTS mess_leave_insert //
CREATE TRIGGER mess_leave_insert AFTER INSERT ON Mess_Leaves FOR EACH ROW BEGIN
 DECLARE rate DECIMAL(10,2);
 SELECT value INTO rate FROM Mess_Settings WHERE setting_name='refund_per_day';
 INSERT INTO Mess_Refunds(bt_no,total_days,refund_amount) VALUES(NEW.bt_no,NEW.days,NEW.days*rate)
 ON DUPLICATE KEY UPDATE total_days=total_days+NEW.days, refund_amount=(total_days+NEW.days)*rate;
END //

DROP TRIGGER IF EXISTS mess_leave_delete //
CREATE TRIGGER mess_leave_delete AFTER DELETE ON Mess_Leaves FOR EACH ROW BEGIN
 DECLARE rate DECIMAL(10,2);
 SELECT value INTO rate FROM Mess_Settings WHERE setting_name='refund_per_day';
 UPDATE Mess_Refunds SET total_days=total_days-OLD.days, refund_amount=(total_days-OLD.days)*rate WHERE bt_no=OLD.bt_no;
END //

DROP TRIGGER IF EXISTS mess_leave_update //
CREATE TRIGGER mess_leave_update AFTER UPDATE ON Mess_Leaves FOR EACH ROW BEGIN
 DECLARE rate DECIMAL(10,2);
 SELECT value INTO rate FROM Mess_Settings WHERE setting_name='refund_per_day';
 UPDATE Mess_Refunds SET total_days=total_days-OLD.days+NEW.days, refund_amount=(total_days-OLD.days+NEW.days)*rate WHERE bt_no=NEW.bt_no;
END //

DROP TRIGGER IF EXISTS check_inventory_before_meal //
CREATE TRIGGER check_inventory_before_meal BEFORE INSERT ON Meal_Log FOR EACH ROW BEGIN
 DECLARE done INT DEFAULT 0; DECLARE v_item INT; DECLARE v_req DECIMAL(10,2); DECLARE v_have DECIMAL(10,2);
 DECLARE cur CURSOR FOR SELECT item_id,required_qty FROM Dish_Requirements WHERE dish_id=NEW.dish_id;
 DECLARE CONTINUE HANDLER FOR NOT FOUND SET done=1;
 OPEN cur;
 read_loop: LOOP
  FETCH cur INTO v_item,v_req;
  IF done=1 THEN LEAVE read_loop; END IF;
  SELECT quantity INTO v_have FROM Inventory WHERE item_id=v_item;
  IF v_have < v_req THEN SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='Insufficient inventory stock'; END IF;
 END LOOP;
 CLOSE cur;
END //

DROP TRIGGER IF EXISTS deduct_inventory_after_meal //
CREATE TRIGGER deduct_inventory_after_meal AFTER INSERT ON Meal_Log FOR EACH ROW BEGIN
 UPDATE Inventory i JOIN Dish_Requirements d ON i.item_id=d.item_id
 SET i.quantity=i.quantity-d.required_qty, i.total_cost=i.quantity*i.cost_per_unit
 WHERE d.dish_id=NEW.dish_id;
END //
DELIMITER ;

CREATE TABLE IF NOT EXISTS Vendors(
 vendor_id INT AUTO_INCREMENT PRIMARY KEY,
 vendor_name VARCHAR(100), phone VARCHAR(15), item_speciality VARCHAR(100));

CREATE TABLE IF NOT EXISTS Purchase_Orders(
 po_id INT AUTO_INCREMENT PRIMARY KEY,
 vendor_id INT, order_date DATE, status VARCHAR(20),
 FOREIGN KEY(vendor_id) REFERENCES Vendors(vendor_id));

CREATE TABLE IF NOT EXISTS Purchase_Items(
 po_item_id INT AUTO_INCREMENT PRIMARY KEY,
 po_id INT, item_id INT, qty DECIMAL(10,2), price DECIMAL(10,2),
 FOREIGN KEY(po_id) REFERENCES Purchase_Orders(po_id) ON DELETE CASCADE,
 FOREIGN KEY(item_id) REFERENCES Inventory(item_id));

CREATE TABLE IF NOT EXISTS Waste_Log(
 waste_id INT AUTO_INCREMENT PRIMARY KEY,
 item_id INT, qty DECIMAL(10,2), waste_date DATE, reason VARCHAR(255),
 FOREIGN KEY(item_id) REFERENCES Inventory(item_id));

CREATE TABLE IF NOT EXISTS Mess_Feedback(
 feedback_id INT AUTO_INCREMENT PRIMARY KEY,
 bt_no VARCHAR(20), dish_id INT, rating INT, comment_text VARCHAR(255), feedback_date DATE,
 FOREIGN KEY(bt_no) REFERENCES Students(bt_no) ON DELETE CASCADE,
 FOREIGN KEY(dish_id) REFERENCES Menu_Dishes(dish_id));

CREATE TABLE IF NOT EXISTS Alerts(
 alert_id INT AUTO_INCREMENT PRIMARY KEY,
 alert_type VARCHAR(50), message_text VARCHAR(255), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, is_resolved VARCHAR(10) DEFAULT 'No');

CREATE OR REPLACE VIEW vw_low_inventory AS
SELECT item_id,item_name,quantity,unit FROM Inventory WHERE quantity < 20;

CREATE OR REPLACE VIEW vw_student_refunds AS
SELECT s.bt_no,s.name,r.total_days,r.refund_amount FROM Students s JOIN Mess_Refunds r ON s.bt_no=r.bt_no;

CREATE OR REPLACE VIEW vw_weekly_menu AS
SELECT w.day_name,w.meal_type,m.dish_name FROM Weekly_Menu w JOIN Menu_Dishes m ON w.dish_id=m.dish_id;

DELIMITER //
DROP TRIGGER IF EXISTS low_stock_alert //
CREATE TRIGGER low_stock_alert AFTER UPDATE ON Inventory FOR EACH ROW BEGIN
 IF NEW.quantity < 20 THEN
   INSERT INTO Alerts(alert_type,message_text) VALUES('Low Stock', CONCAT(NEW.item_name,' is below threshold'));
 END IF;
END //
DELIMITER ;


-- ===============================
-- DASHBOARD VIEWS
-- ===============================

CREATE OR REPLACE VIEW vw_monthly_cost_trends AS
SELECT YEAR(expense_date) AS yr, MONTH(expense_date) AS mn, SUM(amount) AS total_expense
FROM Mess_Expenses
GROUP BY YEAR(expense_date), MONTH(expense_date)
ORDER BY yr, mn;

CREATE OR REPLACE VIEW vw_top_waste_items AS
SELECT i.item_name, SUM(w.qty) AS total_wasted
FROM Waste_Log w JOIN Inventory i ON w.item_id=i.item_id
GROUP BY i.item_name
ORDER BY total_wasted DESC;

CREATE OR REPLACE VIEW vw_feedback_average AS
SELECT m.dish_name, ROUND(AVG(f.rating),2) AS avg_rating, COUNT(*) AS total_reviews
FROM Mess_Feedback f JOIN Menu_Dishes m ON f.dish_id=m.dish_id
GROUP BY m.dish_name
ORDER BY avg_rating DESC;
